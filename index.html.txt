 /* ====== 3. TÍNH KẾT QUẢ KHI BẤM NÚT ====== */
         function xemKetQua() {
         var T = 0;            // tổng điểm thô
         var daTraLoi = 0;     // đếm số câu đã trả lời

// cộng điểm 12 câu
        for (var i = 0; i < cauHoi.length; i++) {
             var chon = document.querySelector("input[name='cau" + i + "']:checked");
             if (chon) {
                 T = T + parseInt(chon.value);
                 daTraLoi++;
             }
         }

  // yêu cầu trả lời đủ 12 câu
         if (daTraLoi < 12) {
             alert("Em hãy trả lời đủ 12 câu nhé! (Còn thiếu " + (12 - daTraLoi) + " câu)");
             return;
         }

  // CÔNG THỨC CHÍNH: S = T / 48 * 100
         var S = T / 48 * 100;
         S = Math.round(S * 10) / 10;   // làm tròn 1 chữ số thập phân

  // so sánh với 2 ngưỡng của đề tài: 36,5 và 59,2
         var muc, mau, khuyen;
         if (S < 36.5) {
             muc = "Mức THẤP";
             mau = "#e6f4ea";               // nền xanh nhạt
                 khuyen = "Em đang khá ổn. Hãy tiếp tục giữ thói quen học tập, nghỉ ngơi điều độ và vận động nhẹ mỗi ngày.";
         } else if (S <= 59.2) {
             muc = "Mức TRUNG BÌNH";
             mau = "#fff4e5";               // nền vàng nhạt</p>
                 khuyen = "Em đang có chút căng thẳng. Hãy chia nhỏ mục tiêu học tập, sắp xếp thời gian hợp lí, hít thở sâu và dành thời gian cho sở thích.";
         } else {
             muc = "Mức CAO (cần quan tâm)";
             mau = "#fdecea";               // nền đỏ nhạt
             khuyen = "Em đang khá căng thẳng. Đừng giữ trong lòng - hãy chia sẻ với thầy cô, cha mẹ hoặc chuyên viên tư vấn tâm lí của trường để được hỗ trợ kịp thời.";
         }

  // hiển thị kết quả
         var kq = document.getElementById("ketqua");
          kq.style.display = "block";
          kq.style.background = mau;
          document.getElementById("mucDo").innerText = muc;
           document.getElementById("chiSo").innerText = "Chỉ số căng thẳng của em: S = " + S + " / 100";
          document.getElementById("loiKhuyen").innerText = khuyen;
          kq.scrollIntoView({behavior:"smooth"});
         }
